# 号池调度增强 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 临时故障的渠道自动冷却、按优先级分层尝试、冷却后逐步恢复权重，并在设置页、日志决策记录和两套前端的渠道列表里可见。

**Architecture:** 新配置组 `channel_scheduling_setting` 进入请求策略快照；`service/channel_cooldown.go` 维护内存里的冷却记录，`service/channel_scheduler.go` 在 `CacheGetRandomSatisfiedChannel` 的三处选渠道调用里按四轮规则选渠道；两个开关都关或没开内存缓存时完全走原来的 `model.GetRandomSatisfiedChannel`。前端只读写配置、读冷却状态接口。

**Tech Stack:** Go 1.26（Gin、GORM、testify），React 19 + TypeScript（React Hook Form、Zod、TanStack Query、Vitest），经典版 React + Semi UI。

**Spec:** `接口文档/号池调度增强设计.md`

- 所有改动在独立 worktree `D:/Project/new-api-pool-scheduling`、分支 `feat/pool-scheduling` 上进行，从 `prod` 的 `HEAD` 拉出；只提交到这个本地分支，不 push。主工作区有用户未提交的成本核算改动，禁止在主工作区修改或提交。

- 后端测试用 `github.com/stretchr/testify/require` 和 `assert`；本功能只新增一个后端测试文件 `service/channel_cooldown_test.go`，其余测试扩展已有文件。

- 配置键名、默认值、范围（逐字使用）：
  - `channel_scheduling_setting.cooldown_enabled` bool，默认 `false`
  - `channel_scheduling_setting.cooldown_status_codes` string，默认 `429,500-503,505-523,525-599`
  - `channel_scheduling_setting.cooldown_base_seconds` int，默认 `30`，1–3600
  - `channel_scheduling_setting.cooldown_max_seconds` int，默认 `600`，1–3600，小于首次冷却时运行时按首次冷却封顶
  - `channel_scheduling_setting.disable_after_cooldowns` int，默认 `5`，0–100，0 表示只冷却不禁用
  - `channel_scheduling_setting.disable_error_codes` string，默认空，一行一个，不区分大小写精确匹配
  - `channel_scheduling_setting.tier_attempt_limit` int，默认 `1`，1–100
- 固定禁用原因：`Disabled after repeated temporary failures`（`model.ChannelStatusReasonRepeatedCooldowns`）。
- 决策事件 reason：`priority_limit_reached`、`cooldown_fallback`、`tried_channel_reused`、`session_channel_cooling`、`cooldown_started`、`already_cooling`、`cooldown_disable_requested`、`cooldown_upstream_wide`；source 新增 `cooldown`。

```go
	// doubles the share until it is back to the configured weight.
	cooldownRampStart = 0.25
	// Reaching the cooldown limit does not disable a channel while more than
	// half of a priority level with at least this many channels is cooling
	// down: that points at the upstream rather than at one key.
	cooldownGuardMinTier = 4

type channelFailureClass int

const (
	channelFailureIgnored channelFailureClass = iota
	channelFailurePermanent
	channelFailureTransient
)

// channelCooldown is the in-memory health record of one single-key channel.
type channelCooldown struct {
	// consecutive counts cooldowns without a success in between.
	consecutive int
	// startedAt is the start of the latest cooldown. Results of attempts that
	// began earlier are not counted.
	startedAt time.Time
	// The channel is cooling down while the current time is before until.
	until         time.Time
	weightFactor  float64
	lastStatus    int
	lastErrorCode string
}

// ChannelCooldownStatus is one item of GET /api/channel/cooldowns.
type ChannelCooldownStatus struct {
	ChannelID      int     `json:"channel_id"`
	State          string  `json:"state"`
	CooldownUntil  int64   `json:"cooldown_until"`
	CooldownCount  int     `json:"cooldown_count"`
	WeightFactor   float64 `json:"weight_factor"`
	LastStatusCode int     `json:"last_status_code"`
	LastErrorCode  string  `json:"last_error_code"`
}

var channelCooldowns = struct {
	sync.Mutex
	entries map[int]*channelCooldown
}{entries: make(map[int]*channelCooldown)}

// cooldownNow is the clock of the cooldown records; tests replace it.

// channelCooldownActive reports whether cooldowns take part in channel
// selection. They need the in-memory channel cache as well as the switch.
func channelCooldownActive(policy *model.ChannelSchedulingPolicy) bool {
	return common.MemoryCacheEnabled && policy.CooldownEnabled
}

// classifyChannelFailure sorts a failed attempt into a dead key, a temporary
// upstream failure, or a failure that says nothing about the channel.
func classifyChannelFailure(c *gin.Context, err *types.NewAPIError, policy *model.ChannelSchedulingPolicy) channelFailureClass {
	if err == nil {
		return channelFailureIgnored
	}
	// A client that hung up says nothing about the channel.
	if c != nil && c.Request != nil && c.Request.Context().Err() != nil {
		return channelFailureIgnored
	}
	if !types.IsChannelError(err) {
		if types.IsSkipRetryError(err) {
			return channelFailureIgnored
		}
		if err.GetErrorType() == types.ErrorTypeNewAPIError && err.GetErrorCode() != types.ErrorCodeDoRequestFailed {
			return channelFailureIgnored
		}
	}
	if IsChannelDisableError(err) {
		return channelFailurePermanent
	}
	if err.GetErrorCode() == types.ErrorCodeDoRequestFailed {
		return channelFailureTransient
	}
	if slices.ContainsFunc(policy.CooldownCodes, func(codes operation_setting.StatusCodeRange) bool {
		return err.StatusCode >= codes.Start && err.StatusCode <= codes.End
	}) {
		return channelFailureTransient
	}
	return channelFailureIgnored
}
```

```go
// attemptStartedAt is when the current attempt of the request began.
func attemptStartedAt(state *RequestPolicyState) time.Time {
	if state.AttemptStartedAt.IsZero() {
		return state.StartedAt
	}
	return state.AttemptStartedAt
}

// RecordChannelCooldownFailure counts a failed attempt against the cooldown
// record of a single-key channel. ProcessChannelError calls it before the
// error log is written, so the log carries the cooldown decision.
func RecordChannelCooldownFailure(c *gin.Context, channelError types.ChannelError, err *types.NewAPIError) {
	policy := &model.CurrentRequestPolicy().Scheduling
	if !channelCooldownActive(policy) || channelError.IsMultiKey || channelError.ChannelId <= 0 {
		return
	}
	class := classifyChannelFailure(c, err, policy)
	if class == channelFailureIgnored {
		return
	}
	disableAllowed := common.AutomaticDisableChannelEnabled && channelError.AutoBan
	if class == channelFailurePermanent && disableAllowed {
		// The automatic disable in ProcessChannelError handles a dead key.
		return
	}
	state := RequestPolicy(c)
	attemptStart := attemptStartedAt(state)
	limitEnabled := disableAllowed && policy.DisableAfterCooldowns > 0
	// The priority level of the failed channel decides whether the upstream
	// as a whole is failing.
	var tier []int
	if limitEnabled && c != nil {
		group := common.GetContextKeyString(c, constant.ContextKeyAutoGroup)
		if group == "" {
			group = state.SelectedGroup
		}
		candidates, _ := model.ListSatisfiedChannels(group, c.GetString("original_model"), GetChannelConstraints(c).Filters)
		index := slices.IndexFunc(candidates, func(candidate *model.Channel) bool {
			return candidate.Id == channelError.ChannelId
		})
		for _, candidate := range candidates {
			if index >= 0 && candidate.GetPriority() == candidates[index].GetPriority() {
				tier = append(tier, candidate.Id)
			}
		}
	}

	now := cooldownNow()
	channelCooldowns.Lock()
	entry := channelCooldowns.entries[channelError.ChannelId]
	if entry == nil {
		entry = &channelCooldown{weightFactor: 1}
		channelCooldowns.entries[channelError.ChannelId] = entry
	}
	if now.Before(entry.until) || attemptStart.Before(entry.startedAt) {
		channelCooldowns.Unlock()
		state.AddEvent(PolicyEvent{ChannelID: channelError.ChannelId, Decision: PolicyDecision{Action: "health", Reason: "already_cooling", Source: "cooldown"}})
		return
	}
	entry.consecutive++
	entry.lastStatus, entry.lastErrorCode = err.StatusCode, string(err.GetErrorCode())
	count := entry.consecutive
	limitReached := limitEnabled && count >= policy.DisableAfterCooldowns
	upstreamWide := false
	if limitReached && len(tier) >= cooldownGuardMinTier {
		cooling := 0
		for _, id := range tier {
			other := channelCooldowns.entries[id]
			if id == channelError.ChannelId || other != nil && now.Before(other.until) {
				cooling++
			}
		}
		upstreamWide = cooling*2 > len(tier)
	}
	if limitReached && !upstreamWide {
		// A channel enabled again by the recovery test starts at a quarter of
		// its weight; results of attempts from before the disable are ignored.
		entry.consecutive, entry.until = 0, time.Time{}
		entry.startedAt, entry.weightFactor = now, cooldownRampStart
		channelCooldowns.Unlock()
		state.AddEvent(PolicyEvent{ChannelID: channelError.ChannelId, Decision: PolicyDecision{Action: "health", Reason: "cooldown_disable_requested", Source: "cooldown"}, CooldownCount: count})
		common.SysLog(common.LogText("channel %q (#%d) reached %d consecutive cooldowns, disabling it", channelError.ChannelName, channelError.ChannelId, count))
		gopool.Go(func() {
			DisableChannel(channelError, model.ChannelStatusReasonRepeatedCooldowns)
		})
		return
	}
	duration := policy.CooldownBase
	for range count - 1 {
		if duration >= policy.CooldownMax {
			break
		}
		duration *= 2
	}
	duration = min(duration, policy.CooldownMax)
	entry.startedAt, entry.until, entry.weightFactor = now, now.Add(duration), cooldownRampStart
	channelCooldowns.Unlock()

	seconds := int(duration / time.Second)
	reason := "cooldown_started"
	if limitReached {
		reason = "cooldown_upstream_wide"
		common.SysLog(common.LogText("channel %q (#%d) reached %d consecutive cooldowns, but most channels at its priority are cooling down; keeping it enabled", channelError.ChannelName, channelError.ChannelId, count))
	}
	common.SysLog(common.LogText("channel %q (#%d) cooling down for %ds after %d consecutive temporary failures, last status code %d", channelError.ChannelName, channelError.ChannelId, seconds, count, err.StatusCode))
	state.AddEvent(PolicyEvent{ChannelID: channelError.ChannelId, Decision: PolicyDecision{Action: "health", Reason: reason, Source: "cooldown"}, CooldownCount: count, CooldownSeconds: seconds})
}

// RecordChannelCooldownSuccess lets a channel that succeeded take more traffic
// again: it clears the backoff, ends a cooldown early and doubles the weight
// share of a recovering channel.
func RecordChannelCooldownSuccess(c *gin.Context, channelID int) {
	if channelID <= 0 || !channelCooldownActive(&model.CurrentRequestPolicy().Scheduling) {
		return
	}
	attemptStart := attemptStartedAt(RequestPolicy(c))
	now := cooldownNow()
	channelCooldowns.Lock()
	defer channelCooldowns.Unlock()
	entry := channelCooldowns.entries[channelID]
	if entry == nil || attemptStart.Before(entry.startedAt) {
		return
	}
	entry.consecutive = 0
	if now.Before(entry.until) {
		entry.until = now
	} else {
		entry.weightFactor = min(1, entry.weightFactor*2)
	}
	if entry.weightFactor >= 1 {
		delete(channelCooldowns.entries, channelID)
	}
}

// ClearChannelCooldowns drops the cooldown records of channels an
// administrator edited or switched on or off by hand.
func ClearChannelCooldowns(channelIDs ...int) {
	channelCooldowns.Lock()
	defer channelCooldowns.Unlock()
	for _, id := range channelIDs {
		delete(channelCooldowns.entries, id)
	}
}

// IsChannelCoolingDown reports whether channelID is in an active cooldown.
func IsChannelCoolingDown(channelID int) bool {
	if !channelCooldownActive(&model.CurrentRequestPolicy().Scheduling) {
		return false
	}
	now := cooldownNow()
	channelCooldowns.Lock()
	defer channelCooldowns.Unlock()
	entry := channelCooldowns.entries[channelID]
	return entry != nil && now.Before(entry.until)
}

// ChannelCooldownStatuses lists the cooldown records for the channel list and
// drops the records of channels that no longer exist.
func ChannelCooldownStatuses() []ChannelCooldownStatus {
	now := cooldownNow()
	channelCooldowns.Lock()
	defer channelCooldowns.Unlock()
	statuses := make([]ChannelCooldownStatus, 0, len(channelCooldowns.entries))
	for id, entry := range channelCooldowns.entries {
		if _, err := model.CacheGetChannel(id); err != nil {
			delete(channelCooldowns.entries, id)
			continue
		}
		status := ChannelCooldownStatus{
			ChannelID:      id,
			State:          "recovering",
			CooldownCount:  entry.consecutive,
			WeightFactor:   entry.weightFactor,
			LastStatusCode: entry.lastStatus,
			LastErrorCode:  entry.lastErrorCode,
		}
		if now.Before(entry.until) {
			status.State, status.CooldownUntil = "cooling", entry.until.Unix()
		}
		statuses = append(statuses, status)
	}
	slices.SortFunc(statuses, func(a, b ChannelCooldownStatus) int {
		return cmp.Compare(a.ChannelID, b.ChannelID)
	})
	return statuses
}
```

```go
// schedulerRandFloat draws the weighted channel pick; tests replace it.

// selectScheduledChannel picks the channel of group for one attempt. Without
// the in-memory cache, or with cooldowns off and one attempt per priority
// level, it keeps the priority-per-retry pick of GetRandomSatisfiedChannel.
func selectScheduledChannel(param *RetryParam, group string, priorityRetry int, filters []dto.ChannelFilter) (*model.Channel, error) {
	policy := &model.CurrentRequestPolicy().Scheduling
	if !common.MemoryCacheEnabled || (!policy.CooldownEnabled && policy.TierAttemptLimit <= 1) {
		return model.GetRandomSatisfiedChannel(group, param.ModelName, priorityRetry, filters)
	}
	candidates, err := model.ListSatisfiedChannels(group, param.ModelName, filters)
	if err != nil || len(candidates) == 0 {
		return nil, err
	}
	tried := make(map[int]bool)
	if param.Ctx != nil {
		for _, value := range param.Ctx.GetStringSlice("use_channel") {
			if id, convErr := strconv.Atoi(value); convErr == nil {
				tried[id] = true
			}
		}
	}
	var health map[int]channelCooldown
	if policy.CooldownEnabled {
		health = channelCooldownSnapshot(candidates)
	}
	channel, reason := pickScheduledChannel(candidates, tried, health, cooldownNow(), policy.TierAttemptLimit, schedulerRandFloat)
	if reason != "" && param.Ctx != nil {
		RequestPolicy(param.Ctx).QueueEvent(PolicyEvent{ChannelID: channel.Id, Decision: PolicyDecision{Action: "route", Reason: reason, Source: "routing"}})
	}
	return channel, nil
}

// pickScheduledChannel applies the tiered rounds of the scheduling policy to
// the candidates of one group. tried holds the channels this request already
// attempted and health the cooldown records of the candidates. The reason is
// empty for an ordinary pick and names the fallback otherwise.
func pickScheduledChannel(candidates []*model.Channel, tried map[int]bool, health map[int]channelCooldown, now time.Time, tierLimit int, randFloat func() float64) (*model.Channel, string) {
	sorted := slices.Clone(candidates)
	slices.SortStableFunc(sorted, func(a, b *model.Channel) int {
		return cmp.Compare(b.GetPriority(), a.GetPriority())
	})
	var tiers [][]*model.Channel
	for index, channel := range sorted {
		if index == 0 || channel.GetPriority() != sorted[index-1].GetPriority() {
			tiers = append(tiers, nil)
		}
		tiers[len(tiers)-1] = append(tiers[len(tiers)-1], channel)
	}
	coolingUntil := func(channel *model.Channel) (time.Time, bool) {
		entry, ok := health[channel.Id]
		return entry.until, ok && now.Before(entry.until)
	}

	// Rounds 1 and 2 take untried channels that are not cooling down, highest
	// priority first. Round 1 also skips a priority level other than the
	// lowest once this request tried tierLimit of its channels.
	for _, honorLimit := range []bool{true, false} {
		limitSkipped := false
		for index, tier := range tiers {
			if honorLimit && index < len(tiers)-1 {
				attempted := 0
				for _, channel := range tier {
					if tried[channel.Id] {
						attempted++
					}
				}
				if attempted >= tierLimit {
					limitSkipped = true
					continue
				}
			}
			pool := slices.DeleteFunc(slices.Clone(tier), func(channel *model.Channel) bool {
				_, cooling := coolingUntil(channel)
				return tried[channel.Id] || cooling
			})
			if len(pool) == 0 {
				continue
			}
			reason := ""
			if limitSkipped {
				reason = "priority_limit_reached"
			}
			return pickWeightedChannel(pool, health, randFloat), reason
		}
	}

	// Round 3 takes the untried channel whose cooldown ends first.
	var earliest *model.Channel
	var earliestUntil time.Time
	for _, channel := range sorted {
		until, cooling := coolingUntil(channel)
		if !tried[channel.Id] && cooling && (earliest == nil || until.Before(earliestUntil)) {
			earliest, earliestUntil = channel, until
		}
	}
	if earliest != nil {
		return earliest, "cooldown_fallback"
	}

	// Round 4 reuses a tried channel: the highest priority level with a
	// channel that is not cooling down, else the cooldown that ends first.
	for _, tier := range tiers {
		pool := slices.DeleteFunc(slices.Clone(tier), func(channel *model.Channel) bool {
			_, cooling := coolingUntil(channel)
			return cooling
		})
		if len(pool) > 0 {
			return pickWeightedChannel(pool, health, randFloat), "tried_channel_reused"
		}
	}
	for _, channel := range sorted {
		until, _ := coolingUntil(channel)
		if earliest == nil || until.Before(earliestUntil) {
			earliest, earliestUntil = channel, until
		}
	}
	return earliest, "tried_channel_reused"
}

// pickWeightedChannel draws a channel by its configured weight times its
// cooldown ramp share. As in GetRandomSatisfiedChannel, a channel of weight 0
// only runs when every weight in the pool is 0.
func pickWeightedChannel(pool []*model.Channel, health map[int]channelCooldown, randFloat func() float64) *model.Channel {
	weights := make([]float64, len(pool))
	total := 0.0
	for index, channel := range pool {
		weight := float64(channel.GetWeight())
		if entry, ok := health[channel.Id]; ok {
			weight *= entry.weightFactor
		}
		weights[index] = weight
		total += weight
	}
	if total <= 0 {
		return pool[min(int(randFloat()*float64(len(pool))), len(pool)-1)]
	}
	target := randFloat() * total
	for index, weight := range weights {
		if target < weight {
			return pool[index]
		}
		target -= weight
	}
	// Rounding can leave the target at the very end of the range.
	for index := len(pool) - 1; index >= 0; index-- {
		if weights[index] > 0 {
			return pool[index]
		}
	}
	return pool[len(pool)-1]
}
```

```go
}
```

```js
// Appends new translation keys to the end of the "translation" object of each
// locale file and leaves every other byte unchanged.
// Usage: node append-locale-keys.mjs <locales dir> <data json>
import fs from 'node:fs'
import path from 'node:path'

const [, , localesDir, dataFile] = process.argv
const data = JSON.parse(fs.readFileSync(dataFile, 'utf8'))
const tail = '\n  }\n}\n'
for (const [locale, entries] of Object.entries(data)) {
  const file = path.join(localesDir, `${locale}.json`)
  const text = fs.readFileSync(file, 'utf8')
  if (!text.endsWith(tail)) throw new Error(`${file}: unexpected ending`)
  const existing = JSON.parse(text).translation
  const additions = Object.entries(entries).filter(
    ([key]) => !Object.hasOwn(existing, key)
  )
  if (additions.length > 0) {
    const lines = additions
      .map(([key, value]) => `    ${JSON.stringify(key)}: ${JSON.stringify(value)}`)
      .join(',\n')
    fs.writeFileSync(file, `${text.slice(0, -tail.length)},\n${lines}${tail}`)
    JSON.parse(fs.readFileSync(file, 'utf8'))
  }
  console.log(`${locale}: ${additions.length} added, ${Object.keys(entries).length - additions.length} already present`)
}
```
